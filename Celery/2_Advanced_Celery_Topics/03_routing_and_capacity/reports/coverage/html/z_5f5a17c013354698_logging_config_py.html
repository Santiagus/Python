1"""Centralized logging configuration and pretty development formatting. 

2 

3Provides ContextVar-backed correlation ID propagation and a clean, concise 

4development formatter that highlights timestamps, log levels, truncated request IDs, 

5and domain attributes without cluttering the console with noisy HTTP headers. 

6""" 

7 

8from __future__ import annotations 

9 

10import logging 

11import sys 

12from contextvars import ContextVar 

13 

14# Global context variable for cross-correlating async calls and Celery task traces 

15current_request_id: ContextVar[str | None] = ContextVar("current_request_id", default=None) 

16 

17 

18class RequestContextFilter(logging.Filter): 

19 """Injects the active request_id from ContextVar into every log record.""" 

20 

21 def filter(self, record: logging.LogRecord) -> bool: 

22 """Attach truncated or full request_id to the log record.""" 

23 req_id = getattr(record, "request_id", None) or current_request_id.get() or "-" 

24 record.request_id = req_id 

25 # Truncate request ID to 8 characters for clean console readability 

26 record.req_id_short = req_id[:8] if req_id != "-" else "-" 

27 return True

30class PrettyDevFormatter(logging.Formatter): 

31 """Clean development log formatter. 

32 

33 Formats: 

34 HH:MM:SS [LEVEL] [req_id[:8]] logger_name: message key1=val1 key2=val2 

35 """ 

36 

37 # ANSI color codes for pretty terminal output 

38 COLOR_MAP = { 

39 logging.DEBUG: "\033[36m", # Cyan 

40 logging.INFO: "\033[32m", # Green 

41 logging.WARNING: "\033[33m", # Yellow 

42 logging.ERROR: "\033[31m", # Red 

43 logging.CRITICAL: "\033[35m", # Magenta 

44 } 

45 RESET = "\033[0m" 

46 

47 # Fields to exclude from extra key=value suffix to avoid redundancy 

48 EXCLUDED_RECORD_ATTRS = { 

49 "args", 

50 "asctime", 

51 "created", 

52 "exc_info", 

53 "exc_text", 

54 "filename", 

55 "funcName", 

56 "levelname", 

57 "levelno", 

58 "lineno", 

59 "module", 

60 "msecs", 

61 "msg", 

62 "name", 

63 "pathname", 

64 "process", 

65 "processName", 

66 "relativeCreated", 

67 "stack_info", 

68 "thread", 

69 "threadName", 

70 "request_id", 

71 "req_id_short", 

72 "message", 

73 } 

74 

75 def format(self, record: logging.LogRecord) -> str: 

76 """Format the specified record with truncated UUID and extra attributes.""" 

77 # 1. Format timestamp as HH:MM:SS 

78 record.asctime = self.formatTime(record, datefmt="%H:%M:%S") 

79 

80 # 2. Add color if supported 

81 color = self.COLOR_MAP.get(record.levelno, "") 

82 reset = self.RESET if color else "" 

83 level_str = f"{color}{record.levelname:<5}{reset}" 

84 

85 # 3. Base message prefix: HH:MM:SS [LEVEL] [req_id[:8]] name: message 

86 req_id_short = getattr(record, "req_id_short", "-") 

87 base = f"{record.asctime} [{level_str}] [{req_id_short}] {record.name}: {record.getMessage()}" 

88 

89 # 4. Extract extra keyword attributes formatted as key=value 

90 extras: list[str] = [] 

91 for key, val in record.__dict__.items(): 

92 if key not in self.EXCLUDED_RECORD_ATTRS and not key.startswith("_"): 

93 extras.append(f"{key}={val}") 

94 

95 if extras: 

96 base = f"{base} | {' '.join(extras)}" 

97 

98 # 5. Attach exception traceback if present 

99 if record.exc_info: 

100 if not record.exc_text: 

101 record.exc_text = self.formatException(record.exc_info) 

102 base = f"{base}\n{record.exc_text}" 

103 

104 return base

107def configure_logging(log_level: str = "INFO") -> None: 

108 """Configure the root and application loggers with PrettyDevFormatter. 

109 

110 Args: 

111 log_level: Desired log level string ('DEBUG', 'INFO', 'WARNING', 'ERROR'). 

112 """ 

113 # 1. Determine numeric log level 

114 numeric_level = getattr(logging, log_level.upper(), logging.INFO) 

115 

116 # 2. Create stream handler with context filter and pretty formatter 

117 handler = logging.StreamHandler(sys.stdout) 

118 handler.addFilter(RequestContextFilter()) 

119 handler.setFormatter(PrettyDevFormatter()) 

120 

121 # 3. Configure root logger 

122 root_logger = logging.getLogger() 

123 root_logger.setLevel(numeric_level) 

124 # Clear existing handlers to prevent duplicate lines 

125 root_logger.handlers.clear() 

126 root_logger.addHandler(handler) 

127 

128 # 4. Quiet excessively noisy third-party libraries in development 

129 logging.getLogger("uvicorn.access").setLevel(logging.WARNING) 

130 logging.getLogger("uvicorn.error").setLevel(logging.INFO) 

131 logging.getLogger("asyncio").setLevel(logging.WARNING) 

132 logging.getLogger("kombu").setLevel(logging.WARNING)
