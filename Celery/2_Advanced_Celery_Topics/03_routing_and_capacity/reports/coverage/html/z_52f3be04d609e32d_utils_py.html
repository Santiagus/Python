47def get_worker_loop() -> asyncio.AbstractEventLoop: 

48 """Return the persistent thread-local event loop, initializing it if needed. 

49 

50 In Celery prefork, solo, or test worker processes, reusing a persistent event loop 

51 across task invocations preserves asyncpg connection pools and prevents 'Future attached 

52 to a different loop' errors caused by creating and destroying ephemeral loops. 

53 

54 Returns: 

55 asyncio.AbstractEventLoop: Open, active event loop for the current thread. 

56 """ 

57 loop = getattr(_thread_local, "loop", None) 

58 if loop is None or loop.is_closed(): 

59 loop = asyncio.new_event_loop() 

60 asyncio.set_event_loop(loop) 

61 _thread_local.loop = loop 

62 return loop

74def run_sync(coro: Coroutine[Any, Any, T]) -> T: 

75 """Execute an asynchronous coroutine synchronously, safe across running event loops. 

76 

77 In standard Celery prefork or solo workers, executes on the worker thread's persistent 

78 event loop via loop.run_until_complete(coro). When invoked inside an already running 

79 event loop (such as pytest-asyncio tests), execution is safely delegated to the pooled 

80 background thread pool rather than spinning up and tearing down new executors. 

81 

82 Args: 

83 coro: The async coroutine to execute. 

84 

85 Returns: 

86 T: The return value of the coroutine. 

87 """ 

88 # 1. Inspect whether the calling thread is already executing inside an active event loop 

89 try: 

90 asyncio.get_running_loop() 

91 is_running = True 

92 except RuntimeError: 

93 is_running = False 

94 

95 # 2. When invoked inside an active loop (e.g. pytest-asyncio), reuse the shared thread pool 

96 if is_running: 

97 executor = get_sync_executor() 

98 return cast(T, executor.submit(cast(Any, run_sync), coro).result()) 

99 

100 # 3. In synchronous worker processes, reuse the persistent thread-local loop 

101 loop = get_worker_loop() 

102 return loop.run_until_complete(coro)
